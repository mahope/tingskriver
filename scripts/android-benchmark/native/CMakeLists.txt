cmake_minimum_required(VERSION 3.22)
project(android_whisper_benchmark LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${WHISPER_SOURCE}/src")
  message(FATAL_ERROR "Set WHISPER_SOURCE to an audited local whisper.cpp tree")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
find_package(Threads REQUIRED)
add_executable(driver driver.cpp)
target_compile_features(driver PRIVATE cxx_std_17)
target_compile_options(driver PRIVATE -Wall -Wextra -Werror)
target_link_libraries(driver PRIVATE whisper Threads::Threads)
